import {
  EMPLOYMENT_TYPES,
  IMPORT_HEADERS,
  type EmploymentTypeValue,
  type ImportRowError,
  type ParsedEmployeeRow,
} from './employee-import.types';

const HEADER_ERROR = 'Header sheet Data tidak sesuai template';
const EMAIL_REGEX = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

export { IMPORT_HEADERS, type ParsedEmployeeRow } from './employee-import.types';

export function assertImportHeaders(header: unknown[]): void {
  for (let i = 0; i < IMPORT_HEADERS.length; i++) {
    const cell = header[i];
    const value = cell == null ? '' : String(cell).trim();
    if (value !== IMPORT_HEADERS[i]) {
      throw new Error(HEADER_ERROR);
    }
  }
}

function cellString(value: unknown): string {
  if (value == null) {
    return '';
  }
  if (value instanceof Date) {
    if (Number.isNaN(value.getTime())) {
      return '';
    }
    const year = value.getUTCFullYear();
    const month = String(value.getUTCMonth() + 1).padStart(2, '0');
    const day = String(value.getUTCDate()).padStart(2, '0');
    return `${year}-${month}-${day}`;
  }
  if (typeof value === 'number' && Number.isFinite(value)) {
    const digits = Number.isSafeInteger(value)
      ? value.toLocaleString('fullwide', { useGrouping: false })
      : String(value);
    if (/^8\d{8,12}$/.test(digits)) {
      return `0${digits}`;
    }
    return digits;
  }
  if (typeof value === 'object') {
    const record = value as {
      text?: unknown;
      result?: unknown;
      richText?: { text?: string }[];
    };
    if (typeof record.text === 'string') {
      return record.text.trim();
    }
    if (Array.isArray(record.richText)) {
      return record.richText
        .map((part) => part.text ?? '')
        .join('')
        .trim();
    }
    if ('result' in record) {
      return cellString(record.result);
    }
  }
  return String(value).trim();
}

function isBlankRow(cells: string[]): boolean {
  return cells.every((c) => c === '');
}

function findDepartmentId(
  raw: string,
  departments: { id: string; name: string; code: string }[],
): string | null {
  if (raw === '') {
    return null;
  }
  const byCode = departments.find((d) => d.code === raw);
  if (byCode) {
    return byCode.id;
  }
  const lower = raw.toLowerCase();
  const byName = departments.find((d) => d.name.trim().toLowerCase() === lower);
  return byName?.id ?? null;
}

function parseEmploymentType(raw: string): EmploymentTypeValue | null {
  if (raw === '') {
    return null;
  }
  if ((EMPLOYMENT_TYPES as readonly string[]).includes(raw)) {
    return raw as EmploymentTypeValue;
  }
  return null;
}

function parseJoinDate(raw: string): string | null | 'invalid' {
  if (raw === '') {
    return null;
  }
  const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(raw);
  if (!match) {
    return 'invalid';
  }
  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = Number(match[3]);
  const utc = Date.UTC(year, month - 1, day);
  const d = new Date(utc);
  if (
    d.getUTCFullYear() !== year ||
    d.getUTCMonth() !== month - 1 ||
    d.getUTCDate() !== day
  ) {
    return 'invalid';
  }
  return raw;
}

export function jakartaJoinDate(now: Date = new Date()): Date {
  const formatter = new Intl.DateTimeFormat('en-CA', {
    timeZone: 'Asia/Jakarta',
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
  });
  const parts = formatter.formatToParts(now);
  const year = parts.find((p) => p.type === 'year')?.value ?? '1970';
  const month = parts.find((p) => p.type === 'month')?.value ?? '01';
  const day = parts.find((p) => p.type === 'day')?.value ?? '01';
  return new Date(`${year}-${month}-${day}T00:00:00.000Z`);
}

export function parseEmployeeSheet(
  rows: unknown[][],
  departments: { id: string; name: string; code: string }[],
): { rows: ParsedEmployeeRow[]; errors: ImportRowError[] } {
  if (rows.length === 0) {
    throw new Error(HEADER_ERROR);
  }

  assertImportHeaders(rows[0]);

  const parsedRows: ParsedEmployeeRow[] = [];
  const errors: ImportRowError[] = [];
  const seenEmails = new Set<string>();

  for (let i = 1; i < rows.length; i++) {
    const sheetRow = i + 1;
    const rawRow = rows[i] ?? [];
    const cells = IMPORT_HEADERS.map((_, col) => cellString(rawRow[col]));

    if (isBlankRow(cells)) {
      continue;
    }

    const messages: string[] = [];

    const firstName = cells[0];
    const lastName = cells[1];
    const departmentRaw = cells[2];
    const employmentRaw = cells[3];
    const emailRaw = cells[4];
    const phoneRaw = cells[5];
    const joinDateRaw = cells[6];

    if (firstName === '') {
      messages.push('Nama depan wajib diisi');
    }
    if (lastName === '') {
      messages.push('Nama belakang wajib diisi');
    }

    const departmentId = findDepartmentId(departmentRaw, departments);
    if (departmentId == null) {
      messages.push('Departemen tidak ditemukan');
    }

    const employmentType = parseEmploymentType(employmentRaw);
    if (employmentType == null) {
      messages.push('Tipe kerja tidak valid');
    }

    let email: string | null = null;
    if (emailRaw === '') {
      email = null;
    } else if (!EMAIL_REGEX.test(emailRaw)) {
      messages.push('Email tidak valid');
    } else {
      const normalized = emailRaw.toLowerCase();
      if (seenEmails.has(normalized)) {
        messages.push('Email sudah dipakai');
      } else {
        email = emailRaw;
      }
    }

    const phone = phoneRaw === '' ? null : phoneRaw;

    const joinDateResult = parseJoinDate(joinDateRaw);
    let joinDate: string | null = null;
    if (joinDateResult === 'invalid') {
      messages.push('Tanggal bergabung tidak valid');
    } else {
      joinDate = joinDateResult;
    }

    if (messages.length > 0) {
      errors.push({ row: sheetRow, message: messages.join('; ') });
      continue;
    }

    parsedRows.push({
      row: sheetRow,
      firstName,
      lastName,
      departmentId: departmentId!,
      employmentType: employmentType!,
      email,
      phone,
      joinDate,
    });
    if (email != null) {
      seenEmails.add(email.toLowerCase());
    }
  }

  return { rows: parsedRows, errors };
}

import * as ExcelJS from 'exceljs';
import { IMPORT_HEADERS } from './employee-import.types';

export async function buildEmployeeImportTemplate(): Promise<Buffer> {
  const workbook = new ExcelJS.Workbook();

  const dataSheet = workbook.addWorksheet('Data');
  dataSheet.addRow([...IMPORT_HEADERS]);

  const sampleSheet = workbook.addWorksheet('Contoh');
  sampleSheet.addRow([...IMPORT_HEADERS]);
  sampleSheet.addRow([
    'Amir',
    'Mufid',
    'Engineering',
    'PERMANENT',
    'amir@eiger.test',
    '08123456789',
    '2026-01-15',
  ]);

  const xlsxBuffer = await workbook.xlsx.writeBuffer();
  return Buffer.from(xlsxBuffer) as Buffer;
}

function isBlankDataRow(cells: string[]): boolean {
  return cells.every((c) => c === '');
}

async function countNonEmptyDataRows(buffer: Buffer): Promise<number> {
  const workbook = new ExcelJS.Workbook();
  try {
    await workbook.xlsx.load(
      buffer as unknown as Awaited<
        ReturnType<InstanceType<typeof ExcelJS.Workbook>['xlsx']['writeBuffer']>
      >,
    );
  } catch {
    return 0;
  }
  const sheet = workbook.getWorksheet('Data');
  if (!sheet) {
    return 0;
  }
  let count = 0;
  sheet.eachRow((row, rowNumber) => {
    if (rowNumber === 1) {
      return;
    }
    const values = row.values;
    const rawRow = Array.isArray(values) ? values.slice(1) : [];
    const cells = IMPORT_HEADERS.map((_, col) => {
      const cell = rawRow[col];
      return cell == null ? '' : String(cell).trim();
    });
    if (!isBlankDataRow(cells)) {
      count++;
    }
  });
  return count;
}

export class EmployeeImportController {
  constructor(

  @Get('template')
  @ApiOperation({ summary: 'Download employee import spreadsheet template' })

  async downloadTemplate(@Res() res: Response): Promise<void> {
    const buffer = await buildEmployeeImportTemplate();
    res.set({
      'Content-Type':
        'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
      'Content-Disposition': 'attachment; filename="template-import-karyawan.xlsx"',
      'Content-Length': buffer.length,
    });
    res.end(buffer);
  }

  @Post()
  @ApiOperation({ summary: 'Enqueue employee spreadsheet import' })
  @RequirePermissions('employees:create')
  @UseInterceptors(
    FileInterceptor('file', {
      limits: { fileSize: MAX_FILE_BYTES },
    }),
  )
  async enqueue(
    @CurrentUser() user: AuthenticatedUser,
    @Body() body: { nikPrefix: string },
    @UploadedFile() file: Express.Multer.File,
  ): Promise<{ jobId: string }> {
    if (!file?.buffer?.length) {
      throw new BadRequestException('File impor wajib diunggah');
    }

    const originalName = file.originalname ?? '';
    if (!/\.xlsx$/i.test(originalName)) {
      throw new BadRequestException('File harus berformat .xlsx');
    }

    let nikPrefix: string;
    try {
      nikPrefix = assertNikPrefix(body.nikPrefix ?? '');
    } catch (err) {
      throw new BadRequestException(
        err instanceof Error ? err.message : String(err),
      );
    }

    const waitingOrActive = await this.queue.getJobs(['waiting', 'active']);
    for (const job of waitingOrActive) {
      if (job.data?.tenantId !== user.tenantId) {
        continue;
      }
      const state = await job.getState();
      const progress = normalizeImportProgress(job.progress);
      throw new ConflictException({ jobId: String(job.id), state, progress });
    }

    const dir = path.join(os.tmpdir(), 'hrms-employee-import');
    await mkdir(dir, { recursive: true });
    const filePath = path.join(dir, `${randomUUID()}.xlsx`);
    await writeFile(filePath, file.buffer);

    const rowCount = await countNonEmptyDataRows(file.buffer);
    if (rowCount > MAX_DATA_ROWS) {
      await unlink(filePath).catch(() => undefined);
      throw new BadRequestException(
        `Maksimal ${MAX_DATA_ROWS} baris data per file impor`,
      );
    }

    const job = await this.queue.add('import', {
      tenantId: user.tenantId,
      filePath,
      nikPrefix,
    });

    return { jobId: String(job.id) };
  }

  @Get(':jobId')
  @ApiOperation({ summary: 'Get employee import job status' })
  @RequirePermissions('employees:create')
  async getJobStatus(
    @CurrentUser() user: AuthenticatedUser,
    @Param('jobId') jobId: string,
  ): Promise<{
    jobId: string;
    state: string;
    progress: ReturnType<typeof normalizeImportProgress>;
    failedReason?: string;
  }> {
    const job = await this.queue.getJob(jobId);
    if (!job || job.data?.tenantId !== user.tenantId) {
      throw new NotFoundException('Job impor tidak ditemukan');
    }

    const state = await job.getState();
    if (!['waiting', 'active', 'completed', 'failed'].includes(state)) {
      throw new NotFoundException('Job impor tidak ditemukan');
    }

    return {
      jobId: String(job.id),
      state,
      progress: normalizeImportProgress(job.progress),
      failedReason:
        state === 'failed' ? job.failedReason || 'Impor gagal' : undefined,
    };
  }
}

import { Processor, WorkerHost } from '@nestjs/bullmq';
import { Logger } from '@nestjs/common';
import { Job } from 'bullmq';
import * as ExcelJS from 'exceljs';
import { unlink } from 'fs/promises';
import { PrismaService } from '../../../prisma/prisma.service';
import { assertImportHeaders, parseEmployeeSheet } from './employee-import-parser';
import { EmployeeImportService } from './employee-import.service';
import type {
  EmployeeImportJobData,
  EmployeeImportProgress,
} from './employee-import.types';

function worksheetToRows(sheet: ExcelJS.Worksheet): unknown[][] {
  const rows: unknown[][] = [];
  sheet.eachRow((row, rowNumber) => {
    const values = row.values;
    const cells = Array.isArray(values) ? values.slice(1) : [];
    while (rows.length < rowNumber - 1) {
      rows.push([]);
    }
    rows[rowNumber - 1] = cells;
  });
  return rows;
}

@Processor('employee-import')
export class EmployeeImportProcessor extends WorkerHost {
  private readonly logger = new Logger(EmployeeImportProcessor.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly employeeImportService: EmployeeImportService,
  ) {
    super();
  }

  async process(job: Job<EmployeeImportJobData>): Promise<void> {
    const { tenantId, filePath, nikPrefix } = job.data;
    this.logger.debug(`Processing employee import job ${job.id} for tenant ${tenantId}`);

    try {
      const workbook = new ExcelJS.Workbook();
      await workbook.xlsx.readFile(filePath);

      const sheet = workbook.getWorksheet('Data');
      if (!sheet) {
        throw new Error('Sheet Data tidak ditemukan');
      }

      const rows = worksheetToRows(sheet);
      const headerRow = rows[0] ?? [];
      assertImportHeaders(headerRow);

      const departments = await this.prisma.department.findMany({
        where: { tenantId },
        select: { id: true, name: true, code: true },
      });

      const parsed = parseEmployeeSheet(rows, departments);
      const total = parsed.rows.length + parsed.errors.length;

      const afterParse: EmployeeImportProgress = {
        total,
        processed: total,
        created: 0,
        failed: parsed.errors.length,
        errors: [...parsed.errors],
      };
      await job.updateProgress(afterParse);

      const emails = parsed.rows
        .map((row) => row.email)
        .filter((email): email is string => email != null && email !== '');

      const existingRecords =
        emails.length === 0
          ? []
          : await this.prisma.employee.findMany({
              where: { tenantId, email: { in: emails } },
              select: { email: true },
            });

      const existingEmails = new Set(
        existingRecords
          .map((record) => record.email?.toLowerCase())
          .filter((email): email is string => email != null),
      );

      const importResult = await this.employeeImportService.importRows(
        tenantId,
        nikPrefix,
        parsed.rows,
        existingEmails,
      );

      const combinedErrors = [...parsed.errors, ...importResult.errors];
      const afterImport: EmployeeImportProgress = {
        total,
        processed: total,
        created: importResult.created,
        failed: parsed.errors.length + importResult.failed,
        errors: combinedErrors,
      };
      await job.updateProgress(afterImport);
    } catch (error) {
      this.logger.error(
        `Employee import job ${job.id} failed: ${error instanceof Error ? error.message : error}`,
      );
      throw error;
    } finally {
      await unlink(filePath).catch(() => undefined);
    }
  }
}

===== untracked backend/src/modules/employees/import/employee-import.service.ts =====
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../../prisma/prisma.service';
import { jakartaJoinDate } from './employee-import-parser';
import type { ParsedEmployeeRow } from './employee-import.types';
import { nextEmployeeCode } from './nik-sequence';

@Injectable()
export class EmployeeImportService {
  constructor(private readonly prisma: PrismaService) {}

  async importRows(
    tenantId: string,
    prefix: string,
    rows: ParsedEmployeeRow[],
    existingEmails: Set<string>,
  ): Promise<{ created: number; failed: number; errors: { row: number; message: string }[] }> {
    const stored = await this.prisma.employee.findMany({
      where: {
        tenantId,
        employeeCode: { startsWith: prefix },
      },
      select: { employeeCode: true },
    });
    const existingCodes = stored.map((e) => e.employeeCode);

    let created = 0;
    let failed = 0;
    const errors: { row: number; message: string }[] = [];

    for (const row of rows) {
      if (row.email) {
        if (existingEmails.has(row.email)) {
          failed++;
          errors.push({ row: row.row, message: 'Email sudah dipakai' });
          continue;
        }
        const duplicate = await this.prisma.employee.findFirst({
          where: { tenantId, email: row.email },
        });
        if (duplicate) {
          failed++;
          errors.push({ row: row.row, message: 'Email sudah dipakai' });
          continue;
        }
      }

      let employeeCode: string;
      try {
        employeeCode = nextEmployeeCode(prefix, existingCodes);
      } catch (err) {
        failed++;
        errors.push({
          row: row.row,
          message: err instanceof Error ? err.message : String(err),
        });
        continue;
      }

      const joinDate = row.joinDate
        ? new Date(`${row.joinDate}T00:00:00.000Z`)
        : jakartaJoinDate();

      try {
        await this.prisma.employee.create({
          data: {
            tenantId,
            employeeCode,
            firstName: row.firstName,
            lastName: row.lastName,
            email: row.email,
            phone: row.phone,
            departmentId: row.departmentId,
            employmentType: row.employmentType,
            joinDate,
            status: 'ACTIVE',
            ssoUserId: null,
          },
        });
        existingCodes.push(employeeCode);
        created++;
      } catch (err) {
        failed++;
        errors.push({
          row: row.row,
          message: err instanceof Error ? err.message : String(err),
        });
      }
    }

    return { created, failed, errors };
  }
}

  'PERMANENT',
  'CONTRACT',
  'TEMPORARY',
  'INTERN',
  'DAILY_WORKER',
] as const;

export type ParsedEmployeeRow = {
  row: number;
  firstName: string;
  lastName: string;
  departmentId: string;
  employmentType: EmploymentTypeValue;
  email: string | null;

  total: number;
  processed: number;
  created: number;
  failed: number;
  errors: { row: number; message: string }[];
};

export const EMPTY_IMPORT_PROGRESS: EmployeeImportProgress = {
  total: 0,
  processed: 0,
  created: 0,
  failed: 0,
  errors: [],
};

export function normalizeImportProgress(raw: unknown): EmployeeImportProgress {
  if (typeof raw !== 'object' || raw === null) {
    return { ...EMPTY_IMPORT_PROGRESS };
  }
  const value = raw as Partial<EmployeeImportProgress>;
  return {
    total: value.total ?? 0,
    processed: value.processed ?? 0,
    created: value.created ?? 0,
    failed: value.failed ?? 0,
    errors: Array.isArray(value.errors) ? value.errors : [],
  };
}

const PREFIX = /^[1-9]\d$/;

export function assertNikPrefix(prefix: string): string {
  const value = prefix.trim();
  if (!PREFIX.test(value)) {
    throw new Error('Prefix NIK harus 2 digit dari 10 sampai 99');
  }
  return value;
}

export function nextEmployeeCode(prefix: string, existingCodes: string[]): string {
  const normalized = assertNikPrefix(prefix);
  let maxSerial = 0;
  for (const code of existingCodes) {
    if (!code.startsWith(normalized) || !/^\d{6}$/.test(code)) continue;
    maxSerial = Math.max(maxSerial, Number(code.slice(2)));
  }
  const next = maxSerial + 1;
  if (next > 9999) {
    throw new Error('Nomor urut NIK untuk prefix ini sudah habis');
  }
  return `${normalized}${String(next).padStart(4, '0')}`;
}

'use client';

import React, { useCallback, useEffect, useRef, useState } from 'react';
import axios from 'axios';
import { Download, Upload } from 'lucide-react';
import { Button, FormError, Input, Modal, ModalFooter } from '@/components/ui';
import { employeesApi } from '@/lib/api';
import { downloadBlob } from '@/lib/download-blob';

const NIK_PREFIX_REGEX = /^[1-9]\d$/;
const PREFIX_ERROR = 'Prefix NIK harus 2 digit dari 10 sampai 99';

type ImportJobState = 'waiting' | 'active' | 'completed' | 'failed';

interface ImportProgress {
  total: number;
  processed: number;
  created: number;
  failed: number;
  errors: { row: number; message: string }[];
}

interface ImportStatus {
  jobId: string;
  state: ImportJobState;
  progress: ImportProgress;
  failedReason?: string;
}

export interface EmployeeImportDialogProps {
  open: boolean;
  jobId: string;
  onClose: () => void;
  onCompleted: () => void;
}

export function EmployeeImportDialog({
  open,
  jobId: jobIdProp,
  onClose,
  onCompleted,
}: EmployeeImportDialogProps) {
  const [nikPrefix, setNikPrefix] = useState('');
  const [prefixError, setPrefixError] = useState<string | null>(null);
  const [activeJobId, setActiveJobId] = useState('');
  const [status, setStatus] = useState<ImportStatus | null>(null);
  const [uploading, setUploading] = useState(false);
  const [uploadError, setUploadError] = useState<string | null>(null);
  const [downloadingTemplate, setDownloadingTemplate] = useState(false);
  const completedNotifiedRef = useRef(false);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const effectiveJobId = activeJobId || jobIdProp;
  const showProgress = Boolean(effectiveJobId);

  useEffect(() => {
    if (!open) {
      setNikPrefix('');
      setPrefixError(null);
      setActiveJobId('');
      setStatus(null);
      setUploading(false);
      setUploadError(null);
      completedNotifiedRef.current = false;
    }
  }, [open]);

  useEffect(() => {
    if (!open || !effectiveJobId) return;

    let cancelled = false;
    let intervalId: ReturnType<typeof setInterval> | undefined;

    const poll = async () => {
      try {
        const response = await employeesApi.getImportStatus(effectiveJobId);
        if (cancelled) return;
        const data = response.data as ImportStatus;
        setStatus(data);
        setUploadError(null);

        if (data.state === 'completed' || data.state === 'failed') {
          if (intervalId) clearInterval(intervalId);
          if (data.state === 'completed' && !completedNotifiedRef.current) {
            completedNotifiedRef.current = true;
            onCompleted();
          }
        }
      } catch {
        if (!cancelled) {
          setUploadError('Gagal memuat status impor');
        }
      }
    };

    void poll();
    intervalId = setInterval(() => {
      void poll();
    }, 1000);

    return () => {
      cancelled = true;
      if (intervalId) clearInterval(intervalId);
    };
  }, [open, effectiveJobId, onCompleted]);

  const handleDownloadTemplate = async () => {
    setDownloadingTemplate(true);
    try {
      const response = await employeesApi.downloadImportTemplate();
      downloadBlob(response.data as Blob, 'template-import-karyawan.xlsx');
    } catch {
      setUploadError('Gagal mengunduh template');
    } finally {
      setDownloadingTemplate(false);
    }
  };

  const validateAndUpload = useCallback(
    async (file: File) => {
      if (!NIK_PREFIX_REGEX.test(nikPrefix)) {
        setPrefixError(PREFIX_ERROR);
        return;
      }
      setPrefixError(null);
      setUploadError(null);
      setUploading(true);
      try {
        const response = await employeesApi.enqueueImport(file, nikPrefix);
        setActiveJobId(response.data.jobId);
      } catch (error) {
        if (axios.isAxiosError(error) && error.response?.status === 409) {
          const body = error.response.data as { jobId?: string };
          if (body?.jobId) {
            setActiveJobId(body.jobId);
            return;
          }
        }
        setUploadError('Gagal mengunggah file impor');
      } finally {
        setUploading(false);
      }
    },
    [nikPrefix],
  );

  const handleFileChange = (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    event.target.value = '';
    if (!file) return;
    if (!file.name.toLowerCase().endsWith('.xlsx')) {
      setUploadError('File harus berformat .xlsx');
      return;
    }
    void validateAndUpload(file);
  };

  const progress = status?.progress;
  const progressPercent =
    progress && progress.total > 0
      ? Math.min(100, Math.round((progress.processed / progress.total) * 100))
      : 0;

  return (
    <Modal isOpen={open} onClose={onClose} title="Impor karyawan" size="lg">
      {showProgress ? (
        <div className="space-y-4">
          <div>
            <div className="flex justify-between text-sm text-warm-600 mb-1">
              <span>Progress</span>
              <span>
                {progress?.processed ?? 0} / {progress?.total ?? 0}
              </span>
            </div>
            <div className="h-2 bg-warm-100 rounded-full overflow-hidden">
              <div
                className="h-full bg-primary-500 transition-all duration-300"
                style={{ width: `${progressPercent}%` }}
              />
            </div>
          </div>
          <div className="flex gap-4 text-sm">
            <span className="text-green-700 font-medium">{progress?.created ?? 0} berhasil</span>
            <span className="text-red-600 font-medium">{progress?.failed ?? 0} gagal</span>
          </div>
          {uploadError ? <FormError message={uploadError} /> : null}
          {status?.state === 'failed' && status.failedReason ? (
            <p className="text-sm text-red-600">{status.failedReason}</p>
          ) : null}
          <ul className="space-y-1 max-h-48 overflow-y-auto">
            {(progress?.errors ?? []).map((err) => (
              <li key={`${err.row}-${err.message}`} className="text-sm text-warm-700">
                Baris {err.row}: {err.message}
              </li>
            ))}
          </ul>
        </div>
      ) : (
        <div className="space-y-4">
          <p className="text-sm text-warm-600">
            Unduh template, isi data karyawan, lalu unggah file Excel (.xlsx).
          </p>
          <Button
            type="button"
            variant="outline"
            onClick={() => void handleDownloadTemplate()}
            disabled={downloadingTemplate}
          >
            <Download className="h-4 w-4 mr-2" />
            Unduh template
          </Button>
          <div>
            <label className="block text-sm font-medium text-warm-700 mb-1">Prefix NIK (10–99)</label>
            <Input
              value={nikPrefix}
              onChange={(e) => {
                setNikPrefix(e.target.value);
                setPrefixError(null);
              }}
              placeholder="Contoh: 10"
              maxLength={2}
            />
            {prefixError ? <FormError message={prefixError} /> : null}
          </div>
          <div>
            <label className="block text-sm font-medium text-warm-700 mb-1">File Excel</label>
            <input
              ref={fileInputRef}
              type="file"
              accept=".xlsx,application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
              className="block w-full text-sm text-warm-600 file:mr-3 file:py-2 file:px-3 file:rounded-lg file:border-0 file:text-sm file:font-medium file:bg-warm-100 file:text-warm-800 hover:file:bg-warm-200"
              onChange={handleFileChange}
              disabled={uploading}
            />
          </div>
          {uploadError ? <FormError message={uploadError} /> : null}
        </div>
      )}
      <ModalFooter>
        <Button type="button" variant="outline" onClick={onClose}>
          Tutup
        </Button>
        {!showProgress ? (
          <Button
            type="button"
            onClick={() => fileInputRef.current?.click()}
            disabled={uploading}
          >
            <Upload className="h-4 w-4 mr-2" />
            {uploading ? 'Mengunggah…' : 'Pilih file'}
          </Button>
        ) : null}
      </ModalFooter>
    </Modal>
  );
}
